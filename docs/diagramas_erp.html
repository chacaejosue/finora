Table personas {
  id bigint [pk, increment]
  ci varchar [unique]
  nombres varchar
  primer_apellido varchar
  segundo_apellido varchar
  telefono varchar
  direccion varchar
  genero varchar
}

Table usuarios {
  id bigint [pk, increment]
  persona_id bigint [unique, ref: - personas.id]
  username varchar [unique]
  password varchar
  rol varchar
  estado boolean
}

Table clientes {
  id bigint [pk, increment]
  persona_id bigint [unique, ref: - personas.id]
  codigo_cliente varchar [unique]
  activo boolean
}

Table proveedores {
  id bigint [pk, increment]
  razon_social varchar
  nit_rut varchar [unique]
  contacto_nombre varchar
  telefono varchar
  direccion varchar
  activo boolean
}

Table lineas {
  id bigint [pk, increment]
  nombre varchar [unique]
  descripcion varchar
  activa boolean
}

Table categorias {
  id bigint [pk, increment]
  nombre varchar [unique]
  descripcion varchar
  activa boolean
}

Table productos {
  id bigint [pk, increment]
  codigo varchar [unique]
  nombre varchar
  descripcion text
  precio_costo decimal
  precio_venta decimal
  stock_actual int
  stock_minimo int
  proveedor_id bigint [ref: > proveedores.id]
  linea_id bigint [null, ref: > lineas.id]
  categoria_id bigint [ref: > categorias.id]
  activo boolean
}

Table compras {
  id bigint [pk, increment]
  numero_factura_nota varchar
  fecha_compra date
  total decimal
  estado varchar
  proveedor_id bigint [ref: > proveedores.id]
  registrado_por_usuario_id bigint [null, ref: > usuarios.id]
}

Table detalle_compras {
  id bigint [pk, increment]
  compra_id bigint [ref: > compras.id]
  producto_id bigint [ref: > productos.id]
  cantidad int
  precio_unitario decimal
  subtotal decimal
}

Table pedidos {
  id bigint [pk, increment]
  codigo varchar [unique]
  cliente_id bigint [ref: > clientes.id]
  registrado_por_usuario_id bigint [null, ref: > usuarios.id]
  fecha_pedido date
  fecha_entrega timestamp
  entregado_por_usuario_id bigint [null, ref: > usuarios.id]
  estado varchar
  total decimal
  direccion_entrega varchar
  costo_delivery decimal
}

Table detalle_pedidos {
  id bigint [pk, increment]
  pedido_id bigint [ref: > pedidos.id]
  producto_id bigint [ref: > productos.id]
  cantidad int
  precio_unitario decimal
  subtotal decimal
}

Table ventas {
  id bigint [pk, increment]
  numero_comprobante varchar [unique]
  cliente_id bigint [ref: > clientes.id]
  pedido_id bigint [null, ref: - pedidos.id]
  registrado_por_usuario_id bigint [null, ref: > usuarios.id]
  fecha_venta datetime
  subtotal decimal
  descuento decimal
  total decimal
  tipo_pago varchar
  metodo_pago varchar
  numero_cuotas int
  estado varchar
}

Table detalle_ventas {
  id bigint [pk, increment]
  venta_id bigint [ref: > ventas.id]
  producto_id bigint [ref: > productos.id]
  cantidad int
  precio_unitario decimal
  subtotal decimal
}

Table creditos {
  id bigint [pk, increment]
  venta_id bigint [unique, ref: - ventas.id]
  monto_credito decimal
  saldo_pendiente decimal
  numero_cuotas int
  tasa_interes decimal
  estado varchar
}

Table cuotas {
  id bigint [pk, increment]
  credito_id bigint [ref: > creditos.id]
  numero_cuota int
  fecha_vencimiento date
  monto_cuota decimal
  saldo_cuota decimal
  estado varchar
}

Table pagos {
  id bigint [pk, increment]
  venta_id bigint [ref: > ventas.id]
  registrado_por_usuario_id bigint [null, ref: > usuarios.id]
  fecha_pago datetime
  monto decimal
  metodo_pago varchar
  referencia varchar
}

Table aplicaciones_pago {
  id bigint [pk, increment]
  pago_id bigint [ref: > pagos.id]
  cuota_id bigint [ref: > cuotas.id]
  monto_aplicado decimal
}

Table movimientos_inventario {
  id bigint [pk, increment]
  producto_id bigint [ref: > productos.id]
  registrado_por_usuario_id bigint [null, ref: > usuarios.id]
  tipo_movimiento varchar
  cantidad int
  stock_anterior int
  stock_nuevo int
  motivo varchar
  detalle_compra_id bigint [null, ref: > detalle_compras.id]
  detalle_venta_id bigint [null, ref: > detalle_ventas.id]
}

Table auditorias {
  id bigint [pk, increment]
  usuario_id bigint [null, ref: > usuarios.id]
  accion varchar
  modulo varchar
  direccion_ip varchar
  detalles text
}

Table configuraciones {
  id bigint [pk, increment]
  clave varchar [unique]
  valor text
  descripcion varchar
}

Resumen del Ciclo de Vida del Negocio en Finora

  1. Abastecimiento: Se registran compras a proveedores; cada ítem genera automáticamente un movimiento de entrada en el Kardex y actualiza el stock físico disponible.
  2. Comercial (Order-to-Cash): El cliente solicita pedidos por delivery o compra en tienda. La venta descuenta unidades del Kardex e interactúa con el módulo financiero.
  3. Cobranzas & Créditos: En ventas a plazo se genera un cronograma de cuotas amortizables mediante pagos graduales aplicados a cada cuota.
  4. Control y Trazabilidad: Cada movimiento u operación sensible queda vinculado al usuario ejecutor e indexado en el registro de auditoría.
